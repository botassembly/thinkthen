cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../Cargo.toml" version_line REGEX "^version = ")
string(REGEX REPLACE "^version = \"([^\"]+)\"$" "\\1" package_version "${version_line}")
project(ThinkThenC VERSION "${package_version}" LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(THINKTHEN_FFI_LIBRARY "" CACHE FILEPATH "Matching shared native engine from this build")
set(THINKTHEN_FFI_INCLUDE_DIR "" CACHE PATH "Engine ABI header from this build")
if(NOT EXISTS "${THINKTHEN_FFI_LIBRARY}" OR NOT EXISTS "${THINKTHEN_FFI_INCLUDE_DIR}/thinkthen.h")
    message(FATAL_ERROR "Supply the matching native engine and header from this build")
endif()
if(APPLE)
    set(THINKTHEN_FFI_NAME "libthinkthen.${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.dylib")
    set(THINKTHEN_LINK_NAME "libthinkthen.dylib")
    find_program(THINKTHEN_INSTALL_NAME_TOOL install_name_tool REQUIRED)
elseif(UNIX)
    set(THINKTHEN_FFI_NAME "libthinkthen.so.${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}")
    set(THINKTHEN_LINK_NAME "libthinkthen.so")
    find_program(THINKTHEN_PATCHELF patchelf REQUIRED)
else()
    get_filename_component(THINKTHEN_FFI_NAME "${THINKTHEN_FFI_LIBRARY}" NAME)
endif()
configure_package_config_file(cmake/ThinkThenCConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ThinkThenCConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ThinkThenC)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ThinkThenCConfigVersion.cmake"
    VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ThinkThenCConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/ThinkThenCConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ThinkThenC)
install(FILES "${THINKTHEN_FFI_INCLUDE_DIR}/thinkthen.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${THINKTHEN_FFI_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR} RENAME "${THINKTHEN_FFI_NAME}")
configure_file(cmake/install-engine.cmake.in install-engine.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install-engine.cmake")
configure_file(cmake/thinkthen-c.pc.in thinkthen-c.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/thinkthen-c.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES ../../LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/thinkthen-c)
