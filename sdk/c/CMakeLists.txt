cmake_minimum_required(VERSION 3.18)
project(ThinkThenC LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(JANSSON_ROOT "" CACHE PATH "Prepared Jansson prefix")
add_library(ThinkThenJansson STATIC IMPORTED)
set_target_properties(ThinkThenJansson PROPERTIES IMPORTED_LOCATION "${JANSSON_ROOT}/lib/libjansson.a"
    INTERFACE_INCLUDE_DIRECTORIES "${JANSSON_ROOT}/include")
set(THINKTHEN_FFI_LIBRARY "" CACHE FILEPATH "Matching shared native engine from this build")
set(THINKTHEN_FFI_INCLUDE_DIR "" CACHE PATH "Engine ABI header from this build")
if(NOT EXISTS "${THINKTHEN_FFI_LIBRARY}" OR NOT EXISTS "${THINKTHEN_FFI_INCLUDE_DIR}/thinkthen.h")
    message(FATAL_ERROR "Supply the matching native engine and header from this build")
endif()
add_library(ThinkThenEngine SHARED IMPORTED)
set_target_properties(ThinkThenEngine PROPERTIES IMPORTED_LOCATION "${THINKTHEN_FFI_LIBRARY}" IMPORTED_NO_SONAME TRUE)
add_library(thinkthen_c SHARED src/thinkthen_c.c)
add_library(ThinkThen::C ALIAS thinkthen_c)
set_target_properties(thinkthen_c PROPERTIES EXPORT_NAME C C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_options(thinkthen_c PRIVATE -Wall -Wextra -Werror)
target_include_directories(thinkthen_c PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE "${THINKTHEN_FFI_INCLUDE_DIR}")
target_link_libraries(thinkthen_c PUBLIC $<BUILD_INTERFACE:ThinkThenJansson>
    $<INSTALL_INTERFACE:PkgConfig::ThinkThenJansson> PRIVATE ThinkThenEngine)
if(APPLE)
    set_target_properties(thinkthen_c PROPERTIES INSTALL_RPATH "@loader_path")
else()
    set_target_properties(thinkthen_c PROPERTIES INSTALL_RPATH "$ORIGIN")
    # Keep the linked engine alive even when a plugin drops its wrapper handle.
    target_link_options(thinkthen_c PRIVATE -Wl,-z,nodelete)
endif()
install(TARGETS thinkthen_c EXPORT ThinkThenCTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/thinkthen_c.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${THINKTHEN_FFI_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(EXPORT ThinkThenCTargets NAMESPACE ThinkThen:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ThinkThenC)
configure_package_config_file(cmake/ThinkThenCConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ThinkThenCConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ThinkThenC)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ThinkThenCConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ThinkThenC)
configure_file(cmake/thinkthen-c.pc.in thinkthen-c.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/thinkthen-c.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES ../../LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/thinkthen-c)
install(FILES JANSSON-LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/thinkthen-c)
