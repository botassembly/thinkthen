using System.Runtime.CompilerServices;
using System.Text.Json;
using System.Text.Json.Serialization;
using ThinkThen;

// Separate installed-package entry point for PublishAot with warnings as errors.
// It roots only explicit JsonTypeInfo overloads, including collection and toolbox calls.
await using var client = new ThinkThenClient(new()
{
    Backend = Backend.Stub,
    Json = new JsonSerializerOptions { TypeInfoResolver = RecordJson.Default }
});
var value = new InputRecord("aot", "alpha");
var metadata = RecordJson.Default.InputRecord;
var toolbox = AnnotationQuestions.Parse("""{"ready":{"function":"decide","question":"Ready?"}}""");
var decision = await client.DecideAsync("Ready?", value, metadata);
Successful(decision);
var rowMetadata = (System.Text.Json.Serialization.Metadata.JsonTypeInfo<Row<bool, NoDetail>>)RecordJson.Default.GetTypeInfo(typeof(Row<bool, NoDetail>))!;
string json = JsonSerializer.Serialize(decision, rowMetadata);
if (!json.Contains("probability", StringComparison.Ordinal)) throw new Exception("AOT row serialization lost typed fields");
var settings = new ThinkThenOptions { Backend = Backend.Stub, ApiKey = ApiKey.From("synthetic-secret") };
string settingsJson = JsonSerializer.Serialize(settings, RecordJson.Default.ThinkThenOptions);
if (settingsJson.Contains("synthetic-secret", StringComparison.Ordinal)) throw new Exception("AOT settings serialization leaked a key");
if (JsonSerializer.Deserialize(settingsJson, RecordJson.Default.ThinkThenOptions)!.Backend != Backend.Stub)
    throw new Exception("AOT settings admission changed backend");
Successful(await client.ChooseAsync("Which?", value, ["first", "second"], metadata));
Successful(await client.TagAsync("Which?", value, ["label"], metadata));
Successful(await client.ScoreAsync("Urgent?", value, ["routine", "urgent"], metadata));
Successful(await client.FindAsync("Which line?", value, metadata));
Successful(await client.RecognizeAsync(value, ["person"], metadata));
Successful(await client.RelateAsync(value,
    [new RelationRule { Source = "person", Target = "company", Relation = "works_for" }], metadata));
Successful(await client.AnnotateAsync(toolbox, value, metadata));
Successful(await client.Toolbox(toolbox).AnnotateAsync(value, metadata));
int filtered = 0;
await foreach (var item in client.FilterAsync("Ready?", Records(), metadata))
{
    if (!ReferenceEquals(item.Item, value) || item.Row.Status == RowStatus.Failed) throw new Exception("AOT filter lost original record");
    filtered++;
}
if (filtered != 1) throw new Exception("AOT filter dropped a row");
int ranked = 0;
await foreach (var item in client.RankAsync("Urgent?", Records(), ["routine", "urgent"], metadata))
{
    if (!ReferenceEquals(item.Item, value) || item.Row.Status == RowStatus.Failed) throw new Exception("AOT rank lost original record");
    ranked++;
}
if (ranked != 1) throw new Exception("AOT rank dropped a row");
int annotated = 0;
await foreach (var row in client.Toolbox(toolbox).AnnotateStreamAsync(Records(), metadata))
{
    if (row.Position != 1 || row.Status == RowStatus.Failed) throw new Exception("AOT toolbox lost input position");
    annotated++;
}
if (annotated != 1) throw new Exception("AOT toolbox dropped a row");

async IAsyncEnumerable<InputRecord> Records([EnumeratorCancellation] CancellationToken token = default)
{
    token.ThrowIfCancellationRequested();
    yield return value;
    await Task.Yield();
}
static void Successful<A, D>(Row<A, D> row)
{
    if (row.Status == RowStatus.Failed) throw new Exception("AOT metadata overload returned a failed row");
}
internal sealed record InputRecord(string Id, string Body);
[JsonSourceGenerationOptions(PropertyNamingPolicy = JsonKnownNamingPolicy.CamelCase)]
[JsonSerializable(typeof(InputRecord))]
[JsonSerializable(typeof(Row<bool, NoDetail>))]
[JsonSerializable(typeof(ThinkThenOptions))]
internal partial class RecordJson : JsonSerializerContext { }
