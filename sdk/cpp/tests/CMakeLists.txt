cmake_minimum_required(VERSION 3.20)
project(ThinkThenCppConsumer LANGUAGES CXX)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ThinkThenCpp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(consumer consumer.cpp)
target_link_libraries(consumer PRIVATE ThinkThen::Cpp Threads::Threads)
target_compile_options(consumer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
install(TARGETS consumer RUNTIME DESTINATION bin)
