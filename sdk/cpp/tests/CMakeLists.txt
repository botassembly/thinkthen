cmake_minimum_required(VERSION 3.20)
project(ThinkThenCppConsumer LANGUAGES CXX)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ThinkThenCpp CONFIG REQUIRED COMPONENTS Nlohmann)
find_package(Threads REQUIRED)
add_executable(consumer consumer.cpp)
target_link_libraries(consumer PRIVATE ThinkThen::CppNlohmann)
add_executable(typed_consumer typed_consumer.cpp)
add_executable(interface_consumer interface.cpp)
add_executable(ranges_consumer typed_consumer.cpp)
set_target_properties(ranges_consumer PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED TRUE)
add_executable(network_consumer typed_network_consumer.cpp)
add_executable(unknown_code_consumer unknown_code_consumer.cpp)
foreach(target consumer typed_consumer interface_consumer ranges_consumer network_consumer unknown_code_consumer)
    target_link_libraries(${target} PRIVATE ThinkThen::Cpp Threads::Threads)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
install(TARGETS consumer typed_consumer interface_consumer ranges_consumer network_consumer unknown_code_consumer RUNTIME DESTINATION bin)
