cmake_minimum_required(VERSION 3.15)
project(thinkthen_duckdb LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-types.py
    ${CMAKE_CURRENT_BINARY_DIR}/contract_types.hpp RESULT_VARIABLE type_generation_status)
if(NOT type_generation_status EQUAL 0)
    message(FATAL_ERROR "Cannot derive DuckDB types from the engine schema")
endif()
# DuckDB provides these functions when it builds a community extension.
if(COMMAND build_loadable_extension)
    include(${CMAKE_CURRENT_SOURCE_DIR}/community.cmake)
    return()
endif()
foreach(required DUCKDB_SOURCE DUCKDB_STATIC_DIR THINKTHEN_ARCHIVE THINKTHEN_NATIVE_LIBS)
    if(NOT DEFINED ${required})
        message(FATAL_ERROR "Set ${required}")
    endif()
endforeach()
file(GLOB duckdb_static_libraries "${DUCKDB_STATIC_DIR}/*.a")
add_library(thinkthen MODULE src/extension.cpp)
target_include_directories(thinkthen PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_include_directories(thinkthen PRIVATE ${DUCKDB_SOURCE}/src/include ${DUCKDB_SOURCE}/third_party/yyjson/include ../../crates/thinkthen-ffi/include)
target_compile_options(thinkthen PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
if(APPLE)
    target_link_libraries(thinkthen PRIVATE ${THINKTHEN_ARCHIVE} ${duckdb_static_libraries} ${THINKTHEN_NATIVE_LIBS})
    target_link_options(thinkthen PRIVATE "LINKER:-dead_strip")
else()
    target_link_libraries(thinkthen PRIVATE ${THINKTHEN_ARCHIVE} "-Wl,--start-group" ${duckdb_static_libraries} "-Wl,--end-group" ${THINKTHEN_NATIVE_LIBS})
    target_link_options(thinkthen PRIVATE "LINKER:--gc-sections" "LINKER:--exclude-libs,ALL")
endif()
set_target_properties(thinkthen PROPERTIES PREFIX "" SUFFIX ".duckdb_extension" BUILD_WITH_INSTALL_RPATH TRUE)
if(APPLE)
    target_link_options(thinkthen PRIVATE "LINKER:-exported_symbol,_thinkthen_duckdb_cpp_init")
else()
    target_link_options(thinkthen PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()

if(THINKTHEN_TEST_BUILD)
    target_sources(thinkthen PRIVATE tests/routes.c)
    target_include_directories(thinkthen PRIVATE ../../conformance/native)
    target_compile_definitions(thinkthen PRIVATE THINKTHEN_TEST_BUILD THINKTHEN_TEST_FIXTURE)
    add_executable(thinkthen_abi_check tests/abi.cpp)
    foreach(property INCLUDE_DIRECTORIES COMPILE_OPTIONS LINK_LIBRARIES)
        get_target_property(adapter_property thinkthen ${property})
        set_property(TARGET thinkthen_abi_check PROPERTY ${property} "${adapter_property}")
    endforeach()
    add_custom_command(TARGET thinkthen_abi_check POST_BUILD COMMAND thinkthen_abi_check)
endif()
