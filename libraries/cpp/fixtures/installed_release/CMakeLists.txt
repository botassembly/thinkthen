cmake_minimum_required(VERSION 3.18)
project(installed_consumer LANGUAGES CXX)
find_package(thinkthen-cpp CONFIG REQUIRED)
add_executable(consumer main.cpp)
if(THINKTHEN_LINK_STATIC_C)
  target_link_libraries(consumer PRIVATE thinkthen::thinkthen_cpp_static)
else()
  target_link_libraries(consumer PRIVATE thinkthen::thinkthen_cpp_shared)
  get_target_property(_native thinkthen::native_shared IMPORTED_LOCATION)
  get_filename_component(_native_dir "${_native}" DIRECTORY)
  set_target_properties(consumer PROPERTIES BUILD_RPATH "${_native_dir}")
endif()
