cmake_minimum_required(VERSION 3.18)
project(owned_consumer LANGUAGES CXX)
find_package(thinkthen-cpp REQUIRED CONFIG)
add_executable(owned_consumer main.cpp)
if(THINKTHEN_LINK_STATIC_C)
 set(native_target thinkthen::thinkthen_cpp_static)
else()
 set(native_target thinkthen::thinkthen_cpp_shared)
endif()
target_link_libraries(owned_consumer PRIVATE ${native_target})
target_compile_options(owned_consumer PRIVATE -Wall -Wextra -Werror)

add_executable(session_consumer ../session_consumer.cpp)
target_link_libraries(session_consumer PRIVATE ${native_target})
target_compile_options(session_consumer PRIVATE -Wall -Wextra -Werror)
