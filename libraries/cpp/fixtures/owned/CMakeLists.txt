cmake_minimum_required(VERSION 3.18)
project(owned_consumer LANGUAGES CXX)
find_package(thinkthen-cpp REQUIRED CONFIG)
add_executable(owned_consumer main.cpp)
if(THINKTHEN_LINK_STATIC_C)
 set(native_target thinkthen::thinkthen_cpp_static)
else()
 set(native_target thinkthen::thinkthen_cpp_shared)
endif()
target_link_libraries(owned_consumer PRIVATE ${native_target})
target_compile_options(owned_consumer PRIVATE -Wall -Wextra -Werror)

add_executable(session_consumer ../session_consumer.cpp)
target_link_libraries(session_consumer PRIVATE ${native_target})
target_compile_options(session_consumer PRIVATE -Wall -Wextra -Werror)

file(GLOB example_sources "${CMAKE_PREFIX_PATH}/share/thinkthen-cpp/examples/*.cpp")
foreach(source ${example_sources})
 get_filename_component(function ${source} NAME_WE)
 add_executable(example_${function} "${source}")
 target_link_libraries(example_${function} PRIVATE ${native_target})
 target_compile_options(example_${function} PRIVATE -Wall -Wextra -Werror)
 target_compile_options(example_${function} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
 target_link_options(example_${function} PRIVATE -fsanitize=address)
endforeach()

add_executable(schema_projection ../schema_projection.cpp)
target_link_libraries(schema_projection PRIVATE ${native_target})
target_compile_options(schema_projection PRIVATE -Wall -Wextra -Werror)
