git clone https://github.com/ggml-org/llama.cpp.git
cd llama.cpp
git checkout d81235049384534c167caea52b85a694f6103d14
cmake -B build -DCMAKE_BUILD_TYPE=Release -DGGML_METAL=ON
cmake --build build --target llama-server -j 4

model_repo=https://huggingface.co/ggml-org/Clef-Flash-GGUF
model_revision=4a7a08c09bc63baf043b62b5ba89dd67a0357d95
model_file=Clef-Flash-Q4_K_M.gguf
curl -fL -o "$model_file" \
  "$model_repo/resolve/$model_revision/$model_file"

./build/bin/llama-server \
  --model "$model_file" \
  --alias local \
  --host 127.0.0.1 \
  --port 8080 \
  --ctx-size 4096 \
  --batch-size 4096 \
  --ubatch-size 4096 \
  --parallel 1
